CREATE DATABASE IF NOT EXISTS movie_rental_dw;
USE movie_rental_dw;

CREATE TABLE dim_date (
    date_key INT PRIMARY KEY, full_date DATE,
    day_of_week VARCHAR(10), day_number INT,
    month_number INT, month_name VARCHAR(10),
    quarter INT, year INT, is_weekend TINYINT(1)
);

CREATE TABLE dim_customer (
    customer_key INT AUTO_INCREMENT PRIMARY KEY,
    customer_id SMALLINT, full_name VARCHAR(100),
    email VARCHAR(100), city VARCHAR(50),
    country VARCHAR(50), active_status VARCHAR(10)
);

CREATE TABLE dim_film (
    film_key INT AUTO_INCREMENT PRIMARY KEY,
    film_id SMALLINT, title VARCHAR(255),
    description TEXT, release_year YEAR,
    rating VARCHAR(10), rental_duration TINYINT,
    rental_rate DECIMAL(4,2), length_minutes SMALLINT,
    language VARCHAR(50), category VARCHAR(50)
);

CREATE TABLE dim_store (
    store_key INT AUTO_INCREMENT PRIMARY KEY,
    store_id TINYINT, city VARCHAR(50),
    country VARCHAR(50), address VARCHAR(100),
    manager_name VARCHAR(100)
);

CREATE TABLE dim_staff (
    staff_key INT AUTO_INCREMENT PRIMARY KEY,
    staff_id TINYINT, full_name VARCHAR(100),
    email VARCHAR(100), store_id TINYINT
);

CREATE TABLE dim_location (
    location_key INT AUTO_INCREMENT PRIMARY KEY,
    address VARCHAR(100), city VARCHAR(50),
    country VARCHAR(50), postal_code VARCHAR(20)
);

CREATE TABLE fact_rental (
    rental_key INT AUTO_INCREMENT PRIMARY KEY,
    rental_id INT, date_key INT, customer_key INT,
    film_key INT, store_key INT, staff_key INT,
    rental_duration_days INT, expected_duration_days INT,
    late_return_flag TINYINT(1), rental_count INT DEFAULT 1,
    FOREIGN KEY (date_key)     REFERENCES dim_date(date_key),
    FOREIGN KEY (customer_key) REFERENCES dim_customer(customer_key),
    FOREIGN KEY (film_key)     REFERENCES dim_film(film_key),
    FOREIGN KEY (store_key)    REFERENCES dim_store(store_key),
    FOREIGN KEY (staff_key)    REFERENCES dim_staff(staff_key)
);

CREATE TABLE fact_payment (
    payment_key INT AUTO_INCREMENT PRIMARY KEY,
    payment_id SMALLINT, date_key INT,
    customer_key INT, staff_key INT,
    rental_key INT, amount DECIMAL(5,2),
    payment_count INT DEFAULT 1,
    FOREIGN KEY (date_key)     REFERENCES dim_date(date_key),
    FOREIGN KEY (customer_key) REFERENCES dim_customer(customer_key),
    FOREIGN KEY (staff_key)    REFERENCES dim_staff(staff_key),
    FOREIGN KEY (rental_key)   REFERENCES fact_rental(rental_key)
);

CREATE TABLE fact_inventory (
    inventory_key INT AUTO_INCREMENT PRIMARY KEY,
    film_key INT, store_key INT,
    total_copies INT, times_rented INT,
    availability_rate DECIMAL(5,2),
    FOREIGN KEY (film_key)  REFERENCES dim_film(film_key),
    FOREIGN KEY (store_key) REFERENCES dim_store(store_key)
);
